\section{Interval Arithmetic}

\begin{toolkitbox}{Box (Interval) Arithmetic in $\mathbb{R}^n$ - Quick Reference}
\small
\begin{tabular}{l l}
\toprule
\textbf{Name}  & \textbf{Meaning} \\
\midrule
\hyperref[def:box]{Box (product interval)}  & $[a,b] := \{x\in\mathbb{R}^n : a\le x \le b\}$ \\
\hyperref[def:box-contain]{Containment}  & $[a,b]\subseteq[c,d] \iff c\le a \ \land\ b\le d$ \\
\hyperref[def:box-add]{Minkowski sum}  & $[a,b]\oplus[c,d]=[a+c,\; b+d]$ \\
\hyperref[def:box-sub]{Difference}  & $[a,b]\ominus[c,d]=[a-d,\; b-c]$ \\
\hyperref[def:box-smul]{Scalar mult. $(\lambda\ge 0)$}  & $\lambda[a,b]=[\lambda a,\; \lambda b]$ \\
\hyperref[def:box-smul]{Scalar mult. $(\lambda<0)$}  & $\lambda[a,b]=[\lambda b,\; \lambda a]$ \\
\hyperref[def:box-hadamard]{Coordinatewise product}  & $[a,b]\odot[c,d]=\prod_i [\min S_i,\max S_i]$ \\
\hyperref[def:box-hadamard]{Norm bound (Euclidean)}  & $\|x\|_2 \le \max\{\|a\|_2,\|b\|_2\}$ for $x\in[a,b]$ if $0\le a\le b$ \\
\bottomrule
\end{tabular}
\end{toolkitbox}

\begin{toolkitbox}{Interval Arithmetic - Quick Reference}
\small
\begin{tabular}{l l}
\toprule
\textbf{Name}  & \textbf{Meaning} \\
\midrule
\hyperref[def:interval-add]{Addition}        & $[a,b] + [c,d] = [a+c,\, b+d]$           \\
\hyperref[def:interval-sub]{Subtraction}     & $[a,b] - [c,d] = [a-d,\, b-c]$           \\
\hyperref[def:interval-mult]{Multiplication}  & $[a,b] \cdot [c,d]$ (case analysis)      \\
\hyperref[def:interval-contain]{Containment}     & $[a,b] \subseteq [c,d] \iff c\leq a, b\leq d$ \\
\bottomrule
\end{tabular}
\end{toolkitbox}
